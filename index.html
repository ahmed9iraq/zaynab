Information & Data

Data

هي مجموعة الحقائق أو المادة الخام التي تكون على شكل أرقام وحروف ورموز أو أشكال خاصة، وتحتاج للمعالجة لاستخلاص المعنى أو المعلومات منها.

Information

هي حقائق مهمة ومفيدة يتم الحصول عليها كمخرجات نتيجة معالجة البيانات المدخلة، وتكون منظمة ومرتبة وفق سياق معين بحيث تعطي معنى منطقيًا.

ببساطة: Data هي المادة الخام، وبعد تنظيمها ومعالجتها للحصول على معنى مفيد تصبح Information.

مفهوم قواعد البيانات

Database عبارة عن ملف ضخم يمكن فيه تخزين البيانات بشكل منظم ومرتب، ويمكن استرجاع وتعديل هذه المعلومات في أي وقت.

DBMS = Database Management System، وهي برمجيات خاصة يتم من خلالها تخزين البيانات وإدارتها.

RDBMS = نظام إدارة قواعد البيانات العلائقية، وتخزن البيانات فيه في كائنات قاعدة البيانات المسماة الجداول.

المصطلحالمعنى
Databaseقاعدة بيانات تحتوي على جداول مرتبطة ببعضها.
Tableجدول يتكون من أعمدة وصفوف.
Field / Columnحقل أو عمود يمثل معلومة من نوع محدد.
Tuple / Record / Rowسطر في الجدول يمثل مجموعة معلومات.

أمثلة على RDBMS المذكورة في الملزمة: MS SQL ServerMySQLOracleIBM DB2Microsoft Access

أنواع المفاتيح في قواعد البيانات

Keyالفكرة الأساسية
Primary Keyعمود واحد أو أكثر من الجدول يميز السجلات عن بعضها.
Secondary Keyمفتاح يستخدم لأغراض استرجاع البيانات من الجدول.
Foreign Keyحقل مضاف إلى جدول لربطه بجدول آخر ويمثل عمودًا يشير إلى Primary Key.
Candidate Keyحقل أو مجموعة حقول تحمل خصائص المفتاح الرئيسي ويمكن اعتمادها كمفتاح.
Composite Keyمفتاح يتكون من أكثر من صفة عند تعذر الحصول على مفتاح رئيسي مفرد.
ملاحظة مهمة من الملزمة: Primary Key لا يقبل القيم المكررة ولا يقبل NULL.

تجهيز بيئة العمل

الملزمة تذكر استخدام SQL Server على نظام Windows، وتذكر SQL Server Management Studio و Azure Data Studio.

شرح إضافي: في SQL Server Management Studio نستخدم New Query لفتح نافذة كتابة الأوامر، ثم Execute لتنفيذ الاستعلام.

إنشاء قاعدة بيانات — CREATE DATABASE

الأمر CREATE DATABASE يستخدم لإنشاء قاعدة بيانات جديدة.

CREATE DATABASE db_name;

Example: Create a database named dbStudents.

CREATE DATABASE dbStudents;
شرح إضافي: db_name هو المكان الذي نضع فيه اسم قاعدة البيانات التي نريد إنشاءها.
لا يمكن إنشاء أكثر من قاعدة بيانات بنفس الاسم.

حذف قاعدة البيانات — DROP DATABASE

الأمر DROP DATABASE يستخدم لحذف قاعدة البيانات بشكل نهائي.

DROP DATABASE db_name;

Example:

DROP DATABASE dbStudents;
تحذير: عند حذف قاعدة البيانات لا يمكن العودة والتراجع عن هذه الخطوة، لذلك يجب الحذر عند استخدام هذا الأمر.
تسلسل عملي: أنشئ قاعدة → استخدمها → أنشئ الجداول والبيانات. وإذا أردت حذفها نهائيًا استخدم DROP DATABASE.

VARCHAR و NVARCHAR

VARCHAR مناسب للنصوص التي لا تحتاج Unicode، بينما NVARCHAR ضروري عند التعامل مع لغات مثل العربية والصينية والروسية.

CREATE TABLE Users (
 Name NVARCHAR(10)
);

INSERT INTO Users
VALUES (N'مرحبا');

تحديد قاعدة البيانات — USE

الأمر USE يستخدم لتحديد قاعدة البيانات التي نريد التعامل معها.

USE db_name

مثال كامل:

CREATE DATABASE dbStudents;
USE dbStudents;

إنشاء جدول — CREATE TABLE

الأمر CREATE TABLE يستخدم لإنشاء جدول في قاعدة البيانات.

CREATE TABLE table_name (
 column_name_1 data type,
 column_name_2 data type,
 column_name_3 data type
);

Example: Book

CREATE TABLE Book (
 BookID INT,
 Title VARCHAR(100),
 Author VARCHAR(100),
 Price DECIMAL(8,2)
);
كل عمود يحتاج اسمًا ونوع بيانات. لذلك نحدد column_name ثم data type.

PRIMARY KEY — IDENTITY — NOT NULL

Primary Key

CREATE TABLE Book (
 BookID INT PRIMARY KEY,
 Title VARCHAR(100),
 Author VARCHAR(100),
 Price DECIMAL(8,2)
);

IDENTITY(1,1)

CREATE TABLE Book (
 BookID INT IDENTITY(1,1) PRIMARY KEY,
 Title VARCHAR(100),
 Author VARCHAR(100),
 Price DECIMAL(8,2)
);

الترقيم التلقائي يعطي كل صف رقمًا متسلسلًا تلقائيًا، ويبدأ من 1 ويزيد بمقدار 1 في مثال SQL Server.

NOT NULL

CREATE TABLE Book (
 BookID INT IDENTITY(1,1) PRIMARY KEY,
 Title VARCHAR(100) NOT NULL,
 Author VARCHAR(100) NOT NULL,
 Price DECIMAL(8,2) NOT NULL
);

NULL

NULL ليست قيمة بحد ذاتها، وإنما تعني أنه لم يتم إدخال قيمة للحقل، أي أن الحقل فارغ أو غير محدد.

إضافة صف جديد — INSERT INTO

الأمر INSERT INTO يستخدم لإضافة Row أو Record جديد إلى الجدول.

تحديد أسماء الأعمدة

INSERT INTO table_name (column1, column2, column3, ...)
VALUES (value1, value2, value3, ...);

بدون تحديد أسماء الأعمدة

INSERT INTO table_name
VALUES (value1, value2, value3, ...);
عند عدم تحديد أسماء الأعمدة يجب تمرير القيم بنفس ترتيب الأعمدة في الجدول، مع مراعاة الأعمدة التي يسمح لها بـ NULL أو التي تُنشأ تلقائيًا بواسطة IDENTITY.

Multiple Rows

INSERT INTO Book
VALUES
('Advanced SQL ', 'Sara Ahmed', 40.00),
('C++ for Beginners', 'Omar Khalid', 32.75),
('Data Analysis with Python', 'Lina Mohammad', 50.25),
('Machine Learning ', 'Hassan Ali', 65.00);

SELECT — استرجاع البيانات

الأمر SELECT يستخدم لجلب البيانات المخزنة في جداول قواعد البيانات.

SELECT * FROM Book;

ويمكن تحديد أعمدة معينة:

SELECT Title, Price
FROM Book;
* تعني جلب جميع الأعمدة في الجدول.

الشروط — WHERE / AND / OR

WHERE

SELECT *
FROM Book
WHERE Price > 50;

AND

SELECT *
FROM Book
WHERE Price > 35 AND Author = 'Omar Khalid';

OR

SELECT *
FROM Book
WHERE Price > 65 OR Title = 'Advanced SQL';

BETWEEN و IN

BETWEEN

يستخدم كشرط لوجود قيمة محصورة بين قيمتين محددتين.

SELECT * FROM Book
WHERE Price BETWEEN 40 AND 55;

وللنفي:

SELECT Title, Price FROM Book
WHERE Price NOT BETWEEN 50 AND 60;

IN

يسمح بالبحث عن قيمة موجودة ضمن مجموعة قيم محددة.

SELECT *
FROM Book
WHERE Price IN (40, 55, 60);

ORDER BY و DISTINCT

ORDER BY

SELECT * FROM Book
ORDER BY Price ASC;
SELECT * FROM Book
ORDER BY Price DESC;

يمكن ترتيب أكثر من عمود:

SELECT * FROM Book
ORDER BY Author DESC, Price ASC;

DISTINCT

تزيل القيم المكررة من النتيجة.

SELECT DISTINCT Author
FROM Book;

البحث في الأعمدة النصية — LIKE

الأمر LIKE يتيح البحث في الحقول النصية باستخدام Pattern Search، وغالبًا يستخدم مع الرمز %.

SELECT Title, Author, Price
FROM Book
WHERE Author LIKE 'Al%';
